import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2.0.0"; import { sendEmail } from './mailer.ts'; // 初始化 Supabase 客户端（使用 Service Role Key 获得后台修改权限） const supabase = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); serve(async (req) => { // 处理跨域 (CORS) 预检请求 if (req.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', }, }); } try { const { token } = await req.json(); if (!token) { return new Response(JSON.stringify({ error: "Missing token" }), { status: 400, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }, }); } // 1. 验证 Token 并从数据库获取用户信息（包含邮箱 email 与语言偏好 language） const { data: user, error: userError } = await supabase .from('profiles') .select('id, email, language') .eq('verification_token', token) .single(); if (userError || !user) { return new Response(JSON.stringify({ error: "Invalid or expired token" }), { status: 400, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }, }); } // 2. 标记账户为已激活，并清空 verification_token（防止重复提交触发） const { error: updateError } = await supabase .from('profiles') .update({ is_active: true, verification_token: null }) .eq('id', user.id); if (updateError) { throw new Error("Failed to activate user account."); } // 3. 准备电子日记本专属链接（指向正确的 Dairy 路径） const diaryLink = `https://www.rui-palace.com/Dairy?token=${token}`; // 4. 根据用户的语言偏好（'zh' 或 'en'）构建对应的邮件正文 const isChinese = user.language === 'zh'; const subject = isChinese ? "【蕊宫】您的电子日记本专属访问链接" : "[Rui Palace] Your Electronic Diary Access Link"; const htmlBody = isChinese ? `

尊敬的会员：

恭喜！您的账号已成功激活。欢迎开启您的岁时流转之旅。

请点击下方专属链接进入您的电子日记本：

🔗 点击进入电子日记本

提示：此链接包含您的专属访问凭证，请妥善保管，切勿转发他人。

顺祝安康，
蕊宫 Rui Palace 团队

` : `

Dear Member,

Congratulations! Your account has been successfully activated. Welcome to begin your journey through the seasons with Rui Palace.

Please click the link below to access your electronic diary:

🔗 Click to Enter Electronic Diary

Note: This link contains your secure access token. Please keep it safe and do not share it with others.

Best regards,
The Rui Palace Team

`; // 5. 自动触发邮件发送 await sendEmail({ to: user.email, subject: subject, html: htmlBody, }); return new Response( JSON.stringify({ success: true, message: "Account activated and diary link sent." }), { status: 200, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } } ); } catch (err) { console.error("Edge function error:", err); return new Response( JSON.stringify({ error: err.message || "Internal Server Error" }), { status: 500, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } } ); } });
